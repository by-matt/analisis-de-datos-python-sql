    "\n",
    "# Estructura de categorización\n",

    "    # Determinación del modo de participación\n",
    "    if tiene_equipo and (categoria == \"Profesional\" or categoria == \"Semi-profesional\"):\n",

    "    else:\n",
    "        modo = \"Competencia individual\"\n",

    "    # Mostrar resultados\n",
    "    print(\"\\nResultados de la clasificación:\")\n",
