    "## Requerimiento 1 — Probabilidad Teórica (3 puntos)\n",
    "\n",
    "**Definición de eventos:**\n",
    "- **Evento A:** Que un usuario sea de 'Chile'\n",
    "- **Evento B:** Que un usuario tenga suscripción 'Premium'"
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "N = len(poblacion_df)  # tamaño de la población\n",
    "\n",
    "# Definir los eventos como máscaras booleanas\n",
    "evento_A = poblacion_df['pais'] == 'Chile'\n",
    "evento_B = poblacion_df['tiene_premium'] == True\n",
    "\n",
    "# ── P(A): probabilidad de ser de Chile ──────────────────────────────\n",
    "P_A = evento_A.sum() / N\n",
    "\n",
    "# ── P(B): probabilidad de tener Premium ─────────────────────────────\n",
    "P_B = evento_B.sum() / N\n",
    "\n",
    "# ── P(A ∩ B): intersección — Chile Y Premium ─────────────────────────\n",
    "P_A_and_B = (evento_A & evento_B).sum() / N\n",
    "\n",
    "# ── P(A ∪ B): unión — Chile O Premium (o ambos) ─────────────────────\n",
    "P_A_or_B = P_A + P_B - P_A_and_B\n",
    "\n",
    "print('=== Probabilidades teóricas sobre la población (N=1000) ===')\n",
    "print()\n",
    "print(f'Evento A (usuario de Chile):          n = {evento_A.sum():>4}')\n",
    "print(f'Evento B (usuario Premium):           n = {evento_B.sum():>4}')\n",
    "print(f'A ∩ B   (Chile Y Premium):            n = {(evento_A & evento_B).sum():>4}')\n",
    "print(f'A ∪ B   (Chile O Premium):            n = {(evento_A | evento_B).sum():>4}')\n",
    "print()\n",
    "print(f'P(A) = {evento_A.sum()} / {N} = {P_A:.4f}')\n",
    "print(f'P(B) = {evento_B.sum()} / {N} = {P_B:.4f}')\n",
    "print(f'P(A ∩ B) = P(A) + P(B) - P(A ∪ B) ... calculando directamente:')\n",
    "print(f'P(A ∩ B) = {(evento_A & evento_B).sum()} / {N} = {P_A_and_B:.4f}')\n",
    "print(f'P(A ∪ B) = P(A) + P(B) - P(A ∩ B)')\n",
    "print(f'         = {P_A:.4f} + {P_B:.4f} - {P_A_and_B:.4f}')\n",
    "print(f'         = {P_A_or_B:.4f}')"
   ]

    "#    No garantiza representatividad por país, lo que introduce sesgo potencial\n",
    "#    si hay diferencias en la tasa de Premium entre países.\n",
    "#\n",
    "# 3. MUESTRA ESTRATIFICADA (n≈150):\n",
    "#    P(B) = {P_B_estratificado:.4f}. Al asegurar que cada país esté representado\n",

    "#\n",
    "# CONCLUSIÓN:\n",
    "#    El muestreo estratificado produce estimaciones más estables y confiables\n",
