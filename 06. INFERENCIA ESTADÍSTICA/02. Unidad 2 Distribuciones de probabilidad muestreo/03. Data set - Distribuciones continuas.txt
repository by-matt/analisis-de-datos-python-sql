    "from scipy.stats import uniform\n",
    "import numpy as np\n",
    "import matplotlib.pyplot as plt\n",
    "\n",
    "x = np.linspace(0, 10, 100)\n",
    "y = uniform.pdf(x, loc=2, scale=6)    # a=2, b=2+6=8\n",
    "plt.plot(x, y)\n",
    "plt.title(\"Distribución Uniforme Continua\")\n",
    "plt.xlabel(\"x\")\n",
    "plt.ylabel(\"Densidad\")\n",
    "plt.grid(True)\n",
    "plt.show()"
   ]

  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "# Propiedades y cálculos de la uniforme\n",
    "a_u, b_u = 2, 8   # loc=2, scale=6 → [2, 8]\n",
    "media_u  = (a_u + b_u) / 2\n",
    "var_u    = (b_u - a_u)**2 / 12\n",
    "f_x      = 1 / (b_u - a_u)\n",
    "\n",
    "print('=== Distribución Uniforme Continua: [a=2, b=8] ===')\n",
    "print(f'f(x) = 1/(b-a) = 1/{b_u-a_u} = {f_x:.4f}  (constante en [2,8])')\n",
    "print(f'Media  = (a+b)/2 = ({a_u}+{b_u})/2 = {media_u}')\n",
    "print(f'Var(X) = (b-a)²/12 = {(b_u-a_u)**2}/12 = {var_u:.4f}')\n",
    "print()\n",
    "\n",
    "# Calcular probabilidades de intervalos\n",
    "P_3_6 = uniform.cdf(6, loc=a_u, scale=b_u-a_u) - uniform.cdf(3, loc=a_u, scale=b_u-a_u)\n",
    "P_gt7 = 1 - uniform.cdf(7, loc=a_u, scale=b_u-a_u)\n",
    "print(f'P(3 ≤ X ≤ 6) = {P_3_6:.4f}  (área entre 3 y 6)')\n",
    "print(f'P(X > 7)     = {P_gt7:.4f}')\n",
    "print(f'P(X = 5)     = 0.0000  (probabilidad puntual en continua = 0)')"
   ]

   "cell_type": "code",

   "source": [
    "# Regla empírica 68-95-99.7 y cálculos\n",
    "print('=== Distribución Normal Estándar: μ=0, σ=1 ===')\n",
    "for sigma, pct in [(1,68),(2,95),(3,99.7)]:\n",
    "    p = norm.cdf(sigma) - norm.cdf(-sigma)\n",
    "    print(f'P(|X| ≤ {sigma}σ) = P(-{sigma} ≤ X ≤ {sigma}) = {p:.4f} ≈ {pct}%')\n",
    "\n",
    "print()\n",
    "print('Probabilidades clave:')\n",
    "print(f'P(X < 0)      = {norm.cdf(0):.4f}  (exactamente 50% por simetría)')\n",
    "print(f'P(X > 1.645)  = {1-norm.cdf(1.645):.4f}  (cola superior 5%)')\n",
    "print(f'P(X > 1.96)   = {1-norm.cdf(1.96):.4f}  (cola superior 2.5% → IC 95%)')\n",
    "print(f'P(-1.96≤X≤1.96)= {norm.cdf(1.96)-norm.cdf(-1.96):.4f}  (IC 95%)')\n",
    "print()\n",
    "\n",
    "# Normalización (estandarización)\n",
    "mu_real, sigma_real = 72, 8  # ejemplo: calificaciones\n",
    "x_real = 88\n",
    "z = (x_real - mu_real) / sigma_real\n",
    "print(f'Estandarización: X~N({mu_real},{sigma_real}), x={x_real}')\n",
    "print(f'Z = (x-μ)/σ = ({x_real}-{mu_real})/{sigma_real} = {z:.4f}')\n",
    "print(f'P(X ≤ {x_real}) = P(Z ≤ {z:.2f}) = {norm.cdf(z):.4f}')"

    "### 7.1 Código exacto slide 27"
   ]
  },

   "cell_type": "code",
   "execution_count": null,

   "outputs": [],
   "source": [
    "# Código exacto de la presentación — slide 27\n",
    "from scipy.stats import expon\n",
    "\n",
    "x = np.linspace(0, 10, 500)\n",
    "y = expon.pdf(x, scale=2)      # scale = 1/λ = media\n",
    "plt.plot(x, y)\n",
    "plt.title(\"Distribución Exponencial (media=2)\")\n",
    "plt.grid(True)\n",
    "plt.show()"
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "metadata": {},
   "outputs": [],
   "source": [
    "# Propiedades de la exponencial y la propiedad de falta de memoria\n",
    "lam_exp = 0.5   # λ = 0.5 eventos por unidad de tiempo → media = 1/λ = 2\n",
    "media_exp = 1 / lam_exp\n",
    "var_exp   = 1 / lam_exp**2\n",
    "\n",
    "print('=== Distribución Exponencial (λ=0.5, media=2) ===')\n",
    "print(f'f(x) = λ·e^(-λx) = {lam_exp}·e^(-{lam_exp}·x)')\n",
    "print(f'Media = 1/λ = {media_exp}')\n",
    "print(f'Var   = 1/λ² = {var_exp}')\n",
    "print()\n",
    "\n",
    "# Probabilidades clave\n",
    "P_lt2 = expon.cdf(2, scale=media_exp)\n",
    "P_gt5 = 1 - expon.cdf(5, scale=media_exp)\n",
    "print(f'P(X < 2) = {P_lt2:.4f}  (falla antes de la media)')\n",

    "## 📋 Resumen\n",
    "\n",
    "### Las 6 distribuciones continuas\n",
    "\n",
    "| Distribución | Parámetros | E[X] | Var(X) | `scipy.stats` |\n",
    "|-------------|------------|------|--------|---------------|\n",
    "| Uniforme | a, b | (a+b)/2 | (b-a)²/12 | `uniform(loc=a, scale=b-a)` |\n",
    "| Normal | μ, σ | μ | σ² | `norm(loc=μ, scale=σ)` |\n",
    "| Exponencial | λ | 1/λ | 1/λ² | `expon(scale=1/λ)` |\n",
    "| Chi-cuadrado | df | df | 2·df | `chi2(df=k)` |\n",
    "| t de Student | df | 0 (df>1) | df/(df-2) | `t(df=k)` |\n",
    "| F de Fisher | df₁, df₂ | df₂/(df₂-2) | — | `f(dfn=k1, dfd=k2)` |\n",
    "\n",
    "### Árbol de decisión para elegir distribución\n",
    "\n",
    "```\n",
    "¿Cuál es el objetivo?\n",
    "├── Modelar tiempos entre eventos independientes  → Exponencial\n",
    "├── Representar fenómenos simétricos/errores       → Normal\n",
    "├── Sin información previa, todos equiprobables    → Uniforme\n",
    "├── Comparar dos varianzas                         → F de Fisher\n",
    "├── Comparar medias con muestra pequeña (n<30)     → t de Student\n",
    "└── Prueba de independencia o ajuste               → Chi-cuadrado\n",
    "```\n",
    "\n",
    "> 💡 **Cálculo de probabilidades en continuas:** `dist.cdf(b) - dist.cdf(a)` para P(a≤X≤b). Nunca `dist.pdf(x)` como probabilidad — es densidad, no probabilidad.\n",
    "\n",
    "> 💡 **t converge a normal:** a medida que df aumenta, la t de Student se aproxima a la Normal estándar. Para df > 30, la diferencia práctica es mínima."
